   221→   221→function BackToTop() {
   222→   222→  const [visible, setVisible] = React.useState(false);
   223→   223→  React.useEffect(() => {
   224→   224→    const onScroll = () => setVisible(window.scrollY > 700);
   225→   225→    onScroll();
   226→   226→    window.addEventListener("scroll", onScroll, { passive: true });
   227→   227→    return () => window.removeEventListener("scroll", onScroll);
   228→   228→  }, []);
   229→   229→  if (!visible) return null;
   230→   230→  return (
   231→   231→    <button
   232→   232→      type="button"
   233→   233→      onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
   234→   234→      aria-label="Voltar ao topo da página"
   235→   235→      className="animate-fade-in-up fixed bottom-5 right-5 z-40 flex h-11 w-11 items-center justify-center rounded-full border border-frame/70 bg-card text-gold shadow-lg transition-all hover:scale-105 hover:border-gold hover:shadow-[0_0_14px_color-mix(in_srgb,var(--gold)_40%,transparent)] focus-visible:outline-2 focus-visible:outline-gold"
   236→   236→    >
   237→   237→      <ArrowUp className="h-5 w-5" aria-hidden="true" />
   238→   238→    </button>
   239→   239→  );

   245→   245→function ScrollProgress() {
   246→   246→  const [progress, setProgress] = React.useState(0);
   247→   247→  React.useEffect(() => {
   248→   248→    const onScroll = () => {
   249→   249→      const el = document.documentElement;
   250→   250→      const max = el.scrollHeight - el.clientHeight;
   251→   251→      setProgress(max > 0 ? Math.min(100, (el.scrollTop / max) * 100) : 0);
   252→   252→    };
   253→   253→    onScroll();
   254→   254→    window.addEventListener("scroll", onScroll, { passive: true });
   255→   255→    window.addEventListener("resize", onScroll);
   256→   256→    return () => {
   257→   257→      window.removeEventListener("scroll", onScroll);
   258→   258→      window.removeEventListener("resize", onScroll);
   259→   259→    };
   260→   260→  }, []);
   261→   261→  return (
   262→   262→    <div aria-hidden="true" className="pointer-events-none fixed left-0 top-0 z-50 h-[3px] w-full">
   263→   263→      <div
   264→   264→        className="h-full bg-gradient-to-r from-gold-dark via-gold to-gold-bright shadow-[0_0_8px_color-mix(in_srgb,var(--gold)_60%,transparent)]"
   265→   265→        style={{ width: `${progress}%` }}
   266→   266→      />
   267→   267→    </div>
   268→   268→  );
   269→   269→}
