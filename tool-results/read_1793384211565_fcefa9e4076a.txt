   198→   198→function BackToTop() {
   199→   199→  const [visible, setVisible] = React.useState(false);
   200→   200→  React.useEffect(() => {
   201→   201→    const onScroll = () => setVisible(window.scrollY > 700);
   202→   202→    onScroll();
   203→   203→    window.addEventListener("scroll", onScroll, { passive: true });
   204→   204→    return () => window.removeEventListener("scroll", onScroll);
   205→   205→  }, []);
   206→   206→  if (!visible) return null;
   207→   207→  return (
   208→   208→    <button
   209→   209→      type="button"
   210→   210→      onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
   211→   211→      aria-label="Voltar ao topo da página"
   212→   212→      className="animate-fade-in-up fixed bottom-5 right-5 z-40 flex h-11 w-11 items-center justify-center rounded-full border border-frame/70 bg-card text-gold shadow-lg transition-all hover:scale-105 hover:border-gold hover:shadow-[0_0_14px_color-mix(in_srgb,var(--gold)_40%,transparent)] focus-visible:outline-2 focus-visible:outline-gold"
   213→   213→    >
   214→   214→      <ArrowUp className="h-5 w-5" aria-hidden="true" />
   215→   215→    </button>
   216→   216→  );
   217→   217→}
