   218→// ------------------------------------------------------------
   219→// Botão "voltar ao topo" — aparece após rolagem
   220→// ------------------------------------------------------------
   221→function BackToTop() {
   222→  const [visible, setVisible] = React.useState(false);
   223→  React.useEffect(() => {
   224→    const onScroll = () => setVisible(window.scrollY > 700);
   225→    onScroll();
   226→    window.addEventListener("scroll", onScroll, { passive: true });
   227→    return () => window.removeEventListener("scroll", onScroll);
   228→  }, []);
   229→  if (!visible) return null;
   230→  return (
   231→    <button
   232→      type="button"
   233→      onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
   234→      aria-label="Voltar ao topo da página"
   235→      className="animate-fade-in-up fixed bottom-5 right-5 z-40 flex h-11 w-11 items-center justify-center rounded-full border border-frame/70 bg-card text-gold shadow-lg transition-all hover:scale-105 hover:border-gold hover:shadow-[0_0_14px_color-mix(in_srgb,var(--gold)_40%,transparent)] focus-visible:outline-2 focus-visible:outline-gold"
   236→    >
   237→      <ArrowUp className="h-5 w-5" aria-hidden="true" />
   238→    </button>
   239→  );
   240→}

   242→// ------------------------------------------------------------
   243→// Barra de progresso de leitura (topo da página)
   244→// ------------------------------------------------------------
   245→function ScrollProgress() {
   246→  const [progress, setProgress] = React.useState(0);
   247→  React.useEffect(() => {
   248→    const onScroll = () => {
   249→      const el = document.documentElement;
   250→      const max = el.scrollHeight - el.clientHeight;
   251→      setProgress(max > 0 ? Math.min(100, (el.scrollTop / max) * 100) : 0);
   252→    };
   253→    onScroll();
   254→    window.addEventListener("scroll", onScroll, { passive: true });
   255→    window.addEventListener("resize", onScroll);
   256→    return () => {
   257→      window.removeEventListener("scroll", onScroll);
   258→      window.removeEventListener("resize", onScroll);
   259→    };
   260→  }, []);
   261→  return (
   262→    <div aria-hidden="true" className="pointer-events-none fixed left-0 top-0 z-50 h-[3px] w-full">
   263→      <div
   264→        className="h-full bg-gradient-to-r from-gold-dark via-gold to-gold-bright shadow-[0_0_8px_color-mix(in_srgb,var(--gold)_60%,transparent)]"
   265→        style={{ width: `${progress}%` }}
   266→      />
   267→    </div>
   268→  );
   269→}
