   787→   787→// ------------------------------------------------------------
   788→   788→// Gera a sequência de habilidades (níveis 1–18) da prioridade
   789→   789→// Padrão: 1×P1 / 1×P2 / 1×P3, completa P1, depois P2, por fim P3
   790→   790→// ------------------------------------------------------------
   791→   791→export function buildSkillSequence(priority: ("Q" | "W" | "E")[]): ("Q" | "W" | "E" | "R")[] {
   792→   792→  const [p1, p2, p3] = priority;
   793→   793→  const pool: ("Q" | "W" | "E")[] = [p1, p2, p3];
   794→   794→  pool.push(p1, p1, p1, p1); // P1 total = 5 (maximizada ~nível 8)
   795→   795→  pool.push(p2, p2, p2, p2); // P2 total = 5 (maximizada ~nível 13)
   796→   796→  pool.push(p3, p3, p3, p3); // P3 total = 5 (maximizada no 18)
   797→   797→  const result: ("Q" | "W" | "E" | "R")[] = [];
   798→   798→  let idx = 0;
   799→   799→  for (let level = 1; level <= 18; level++) {
   800→   800→    if (level === 6 || level === 11 || level === 16) {
   801→   801→      result.push("R");
   802→   802→    } else {
   803→   803→      result.push(pool[idx] ?? p3);
   804→   804→      idx++;
   805→   805→    }
   806→   806→  }
   807→   807→  return result;
   808→   808→}
