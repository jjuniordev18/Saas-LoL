   206→// ------------------------------------------------------------
   207→// Botão "voltar ao topo" — aparece após rolagem
   208→// ------------------------------------------------------------
   209→function BackToTop() {
   210→  const [visible, setVisible] = React.useState(false);
   211→  React.useEffect(() => {
   212→    const onScroll = () => setVisible(window.scrollY > 700);
   213→    onScroll();
   214→    window.addEventListener("scroll", onScroll, { passive: true });
   215→    return () => window.removeEventListener("scroll", onScroll);
   216→  }, []);
   217→  if (!visible) return null;
   218→  return (
   219→    <button
   220→      type="button"
   221→      onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
   222→      aria-label="Voltar ao topo da página"
   223→      className="animate-fade-in-up fixed bottom-5 right-5 z-40 flex h-11 w-11 items-center justify-center rounded-full border border-frame/70 bg-card text-gold shadow-lg transition-all hover:scale-105 hover:border-gold hover:shadow-[0_0_14px_color-mix(in_srgb,var(--gold)_40%,transparent)] focus-visible:outline-2 focus-visible:outline-gold"
   224→    >
   225→      <ArrowUp className="h-5 w-5" aria-hidden="true" />
   226→    </button>
   227→  );
   228→}

   659→      <BackToTop />
