     6→export type ChampionClass = "lutador" | "tank" | "duelista" | "mago";
     7→export type Difficulty = 1 | 2 | 3;
     8→export type Tier = "S" | "A" | "B";
     9→
    10→export interface ItemInfo {
    11→  name: string;
    12→  cost: number;
    13→  glyph: ItemGlyph;
    14→  category: ItemCategory;
    15→  desc: string;
    16→}
    17→
    18→export type ItemGlyph =
    19→  | "sword"
    20→  | "axe"
    21→  | "shield"
    22→  | "boot"
    23→  | "flask"
    24→  | "heart"
    25→  | "bolt"
    26→  | "bow"
    27→  | "staff"
    28→  | "crown"
    29→  | "clock"
    30→  | "flame"
    31→  | "fist";
    32→
    33→export type ItemCategory = "starter" | "boots" | "ad" | "tank" | "ap";
    34→
    35→export interface ItemRef {
    36→  id: string;
    37→  note?: string;
    38→}
    39→
    40→export type RuneTree = "precision" | "sorcery" | "resolve" | "inspiration" | "domination";
    41→
    42→export interface RuneInfo {
    43→  name: string;
    44→  tree: RuneTree;
    45→  keystone?: boolean;
    46→  desc: string;
    47→}
    48→
    49→export interface RuneSetup {
    50→  primaryTree: RuneTree;
    51→  keystone: string;
    52→  primary: string[];
    53→  secondaryTree: RuneTree;
    54→  secondary: string[];
    55→  shards: string[];
    56→  explanation: string;
    57→}
    58→
    59→export interface SpellInfo {
    60→  name: string;
    61→  glyph: "bolt" | "portal" | "flame" | "ghost";
    62→  desc: string;
    63→}
    64→
    65→export interface Ability {
    66→  key: "P" | "Q" | "W" | "E" | "R";
    67→  name: string;
    68→  desc: string;
    69→}
    70→
    71→export interface Matchup {
    72→  champion: string;
    73→  note: string;
    74→}
    75→
    76→export interface AltBuild {
    77→  name: string;
    78→  when: string;
    79→  items: ItemRef[];
    80→  note?: string;
    81→}
    82→
    83→export interface Champion {
    84→  id: string;
    85→  name: string;
    86→  title: string;
    87→  classes: ChampionClass[];
    88→  difficulty: Difficulty;
    89→  tier: Tier;
    90→  popular?: boolean;
    91→  image: string;
    92→  color: string;
    93→  playstyle: string;
    94→  summary: string;
    95→  strengths: string[];
    96→  weaknesses: string[];
    97→  startItems: ItemRef[];
    98→  boots: ItemRef[];
    99→  coreItems: ItemRef[];
   100→  situationalItems: ItemRef[];
   101→  runes: RuneSetup;
   102→  altRunesNote: string;
   103→  spells: { id: string; note: string }[];
   104→  skillPriority: ("Q" | "W" | "E")[];
   105→  abilities: Ability[];
   106→  comboTips: string[];
   107→  tips: { early: string[]; mid: string[]; late: string[] };
   108→  strongAgainst: Matchup[];
   109→  weakAgainst: Matchup[];
   110→  counterTips: string[];
   111→  altBuilds: AltBuild[];
   112→}

   787→// ------------------------------------------------------------
   788→// Gera a sequência de habilidades (níveis 1–18) da prioridade
   789→// Padrão: 1×P1 / 1×P2 / 1×P3, completa P1, depois P2, por fim P3
   790→// ------------------------------------------------------------
   791→export function buildSkillSequence(priority: ("Q" | "W" | "E")[]): ("Q" | "W" | "E" | "R")[] {
   792→  const [p1, p2, p3] = priority;
   793→  const pool: ("Q" | "W" | "E")[] = [p1, p2, p3];
   794→  pool.push(p1, p1, p1, p1); // P1 total = 5 (maximizada ~nível 8)
   795→  pool.push(p2, p2, p2, p2); // P2 total = 5 (maximizada ~nível 13)
   796→  pool.push(p3, p3, p3, p3); // P3 total = 5 (maximizada no 18)
   797→  const result: ("Q" | "W" | "E" | "R")[] = [];
   798→  let idx = 0;
   799→  for (let level = 1; level <= 18; level++) {
   800→    if (level === 6 || level === 11 || level === 16) {
   801→      result.push("R");
   802→    } else {
   803→      result.push(pool[idx] ?? p3);
   804→      idx++;
   805→    }
   806→  }
   807→  return result;
   808→}
